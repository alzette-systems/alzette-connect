package main

import (
 "crypto/rand"
 "encoding/hex"
 "encoding/json"
 "fmt"
 "net/http"
 "net/http/httptest"
 "os"
 "os/exec"
 "strings"
 "unicode/utf16"

 "github.com/danieljoos/wincred"
)

func main() {
 if len(os.Args)!=2 { os.Exit(2) }
 if err:=run(os.Args[1]); err!=nil { fmt.Fprintln(os.Stderr,err); os.Exit(1) }
}
func run(helper string) error {
 var nonce [16]byte
 if _,err:=rand.Read(nonce[:]);err!=nil{return err}
 id:=fmt.Sprintf("%x-%x-%x-%x-%x",nonce[0:4],nonce[4:6],nonce[6:8],nonce[8:10],nonce[10:16])
 capability:="alp_"+hex.EncodeToString(nonce[:])+"0123456789"
 accepted:=false
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  if r.URL.Path!="/v1/models" || r.Header.Get("Authorization")!="Bearer "+capability { http.Error(w,"rejected",403);return }
  accepted=true; w.WriteHeader(200)
 }))
 defer server.Close()
 data,_:=json.Marshal(map[string]string{"base_url":server.URL+"/v1","capability":capability})
 units:=append(utf16.Encode([]rune(string(data))),0)
 blob:=make([]byte,len(units)*2)
 for i,u:=range units{blob[2*i]=byte(u);blob[2*i+1]=byte(u>>8)}
 credential:=wincred.NewGenericCredential(id+".alzette-connect.claude-launch")
 credential.UserName=id; credential.CredentialBlob=blob
 if err:=credential.Write();err!=nil{return fmt.Errorf("protected fixture write failed")}
 defer credential.Delete()
 output,err:=exec.Command(helper,"--alzette-claude-credential",id).Output()
 if err!=nil || strings.TrimSpace(string(output))!=capability || !accepted{return fmt.Errorf("GUI credential-helper stdout qualification failed")}
 if err=credential.Delete();err!=nil{return fmt.Errorf("protected fixture cleanup failed")}
 if _,err=exec.Command(helper,"--alzette-claude-credential",id).Output();err==nil{return fmt.Errorf("deleted credential was accepted")}
 return json.NewEncoder(os.Stdout).Encode(map[string]any{"gui_helper_stdout_verified":true,"loopback_authorization_verified":true,"credential_removed":true,"missing_credential_rejected":true,"paid_inference":false})
}
